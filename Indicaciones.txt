# Instalación paso a paso (Windows / PowerShell)

# 1. Crear y activar entorno virtual
python -m venv venv
venv\Scripts\activate

# 2. Instalar dependencias normales
pip install python-telegram-bot==20.7 python-dotenv

# 3. Configurar el token en un archivo .env (NO lo escribas en este archivo)
#    Crea .env en la carpeta del proyecto con esta línea:
#    $env:TELEGRAM_TOKEN=tu_token_de_botfather
#
#    Si tu token anterior quedó escrito en algún archivo compartido,
#    revócalo en @BotFather con /revoke y genera uno nuevo.

# 4. Descargar las fotos de las motos (opcional, una sola vez)
python descargar_imagenes.py
#    Atajos: --limite 10   --marca Yamaha   --solo "Yamaha YZF-R1"

# 5. Ejecutar el bot
python bot.py

# 6. Limpiar temporales (opcional)
python limpieza.py            # borra __pycache__ y .pyc/.tmp/.bak
python limpieza.py --simular  # solo muestra lo que borraría

# Si sale "TimedOut" o "ConnectTimeout" no es un error del codigo:
# el bot ya espera 30 s y hace un chequeo previo. Revisa internet,
# VPN/proxy, firewall y la hora de Windows (ver README).
